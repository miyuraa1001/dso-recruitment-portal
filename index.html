1. Masukkan URL Web App Google Apps Script

Tempelkan URL Web App (`https://script.google.com/macros/s/.../exec`) dari Spreadsheet `Kelengkapan PKB DSO` milikmu.

URL berhasil disimpan di browser.

2. Kode Google Apps Script (`Code.gs`) — Sinkron 100% dengan Sel File `Kelengkapan PKB DSO.xlsx`

Upload file `Kelengkapan PKB DSO.xlsx` ke Google Drive → buka sebagai Google Sheets → klik Extensions > Apps Script → tempelkan kode di bawah ini.

// =========================================================================================
// CODE.GS — AUTO-FILL KE SEL ASLI "Kelengkapan PKB DSO.xlsx" & EXPORT .ZIP (BERISI PDF ASLI)
// =========================================================================================

const DB_SHEET_NAME = 'DB_KANDIDAT';

function doPost(e) {
  try {
    const body = JSON.parse(e.postData.contents);
    const action = body.action || 'SUBMIT';

    if (action === 'SUBMIT') {
      return saveCandidate(body.data);
    } else if (action === 'GENERATE_ZIP') {
      return generateZipFromTemplate(body.data, body.hrd || {});
    }
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(DB_SHEET_NAME);
  if (!sheet || sheet.getLastRow() <= 1) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'success', data: [] }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  const rows = sheet.getDataRange().getValues();
  rows.shift();
  const list = rows.map(r => JSON.parse(r[6])).reverse();
  return ContentService.createTextOutput(JSON.stringify({ status: 'success', data: list }))
    .setMimeType(ContentService.MimeType.JSON);
}

// 1. SIMPAN DATA PESERTA KE SHEET "DB_KANDIDAT"
function saveCandidate(d) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(DB_SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(DB_SHEET_NAME);
    sheet.appendRow(['Timestamp', 'Nama_Lengkap', 'Posisi', 'No_KTP', 'No_HP', 'Rekening_Permata', 'JSON_Payload']);
  }
  d.timestamp = new Date().toLocaleString('id-ID');
  sheet.appendRow([d.timestamp, d.nama, d.posisi, "'" + d.ktp, "'" + d.hp, "'" + d.bank_norek, JSON.stringify(d)]);
  return ContentService.createTextOutput(JSON.stringify({ status: 'success' }))
    .setMimeType(ContentService.MimeType.JSON);
}

// 2. ISI SEL TEMPLATE SPREADSHEET ASLI & EXPORT TIAP SHEET JADI PDF KE DALAM 1 FILE .ZIP
function generateZipFromTemplate(d, hrd) {
  const masterSS = SpreadsheetApp.getActiveSpreadsheet();
  const tempFile = DriveApp.getFileById(masterSS.getId()).makeCopy('TEMP_PKB_' + d.nama);
  const tempSS = SpreadsheetApp.openById(tempFile.getId());

  // Helper taruh tanda tangan Base64 ke sel
  function insertSign(sheet, cellA1) {
    if (!d.signature || d.signature.indexOf('base64,') === -1) return;
    const base64 = d.signature.split('base64,')[1];
    const blob = Utilities.newBlob(Utilities.base64Decode(base64), 'image/png', 'ttd.png');
    const range = sheet.getRange(cellA1);
    const img = sheet.insertImage(blob, range.getColumn(), range.getRow());
    img.setWidth(110).setHeight(45);
  }

  // ---------------- A. ISI SHEET "FLK" ----------------
  const flk = tempSS.getSheetByName('FLK');
  if (flk) {
    flk.getRange('Y3').setValue(d.posisi);
    flk.getRange('Y4').setValue(hrd.notes || '');
    flk.getRange('K8').setValue(d.nama);
    flk.getRange('Z8').setValue(d.panggilan);
    flk.getRange('K9').setValue(d.tempat_lahir + ', ' + d.tgl_lahir);
    flk.getRange('Z9').setValue(d.agama);
    flk.getRange('K10').setValue("'" + d.ktp);
    flk.getRange('Z10').setValue(d.jk);
    flk.getRange('K11').setValue("'" + d.hp);
    flk.getRange('K12').setValue(d.alamat);
    flk.getRange('C13').setValue(d.kota);
    flk.getRange('O13').setValue(d.kodepos);

    if (d.status_nikah === 'Tunangan') flk.getRange('Q18').setValue(d.tgl_nikah);
    if (d.status_nikah === 'Menikah') flk.getRange('Q19').setValue(d.tgl_nikah);
    if (d.status_nikah === 'Bercerai') flk.getRange('Q20').setValue(d.tgl_nikah);

    // Susunan Keluarga Inti (Baris 24 - 28)
    (d.keluarga_inti || []).forEach((k, i) => {
      const r = 24 + i;
      if (r <= 28 && k.nama) {
        flk.getRange('E' + r).setValue(k.nama);
        flk.getRange('L' + r).setValue(k.lp);
        flk.getRange('M' + r).setValue(k.ttl);
        flk.getRange('S' + r).setValue(k.pend);
        flk.getRange('Y' + r).setValue(k.pek);
      }
    });

    // Susunan Keluarga Ortu & Saudara (Baris 32 - 40)
    (d.keluarga_ortu || []).forEach((k, i) => {
      const r = 32 + i;
      if (r <= 40 && k.nama) {
        flk.getRange('E' + r).setValue(k.nama);
        flk.getRange('L' + r).setValue(k.lp);
        flk.getRange('M' + r).setValue(k.ttl);
        flk.getRange('S' + r).setValue(k.pend);
        flk.getRange('Y' + r).setValue(k.pek);
      }
    });

    // Pendidikan Formal (Baris 52 - 54: SMA, Dipl, S1)
    (d.pendidikan || []).forEach((p, i) => {
      const r = 52 + i;
      if (r <= 54 && p.nama) {
        flk.getRange('C' + r).setValue(p.nama);
        flk.getRange('I' + r).setValue(p.fak);
        flk.getRange('M' + r).setValue(p.jur);
        flk.getRange('Q' + r).setValue(p.prodi);
        flk.getRange('T' + r).setValue(p.kota);
        flk.getRange('X' + r).setValue(p.thn);
        flk.getRange('AB' + r).setValue(p.ket);
      }
    });

    // Kursus Non Formal (Baris 58 - 59)
    (d.kursus || []).forEach((k, i) => {
      const r = 58 + i;
      if (r <= 59 && k.nama) {
        flk.getRange('B' + r).setValue(k.nama);
        flk.getRange('M' + r).setValue(k.tempat);
        flk.getRange('T' + r).setValue(k.thn);
        flk.getRange('Y' + r).setValue(k.ket);
      }
    });

    // Bahasa Asing (Baris 64 - 65)
    (d.bahasa || []).forEach((b, i) => {
      const r = 64 + i;
      if (r <= 65 && b.nama) {
        flk.getRange('B' + r).setValue(b.nama);
        if (b.lisan === 'Kurang') flk.getRange('M' + r).setValue('V');
        if (b.lisan === 'Cukup') flk.getRange('P' + r).setValue('V');
        if (b.lisan === 'Baik') flk.getRange('S' + r).setValue('V');
        if (b.tulis === 'Kurang') flk.getRange('V' + r).setValue('V');
        if (b.tulis === 'Cukup') flk.getRange('Y' + r).setValue('V');
        if (b.tulis === 'Baik') flk.getRange('AB' + r).setValue('V');
      }
    });

    // Pengalaman Kerja (Baris 70 - 74)
    (d.kerja || []).forEach((pk, i) => {
      const r = 70 + i;
      if (r <= 74 && pk.pt) {
        flk.getRange('B' + r).setValue(pk.pt);
        flk.getRange('J' + r).setValue(pk.jab);
        flk.getRange('O' + r).setValue(pk.thn);
        flk.getRange('T' + r).setValue(pk.gaji);
        flk.getRange('Y' + r).setValue(pk.alasan);
      }
    });

    flk.getRange('O76').setValue(d.pt_senang);
    flk.getRange('E77').setValue(d.alasan_senang);

    // Prioritas Minat (O83:O87 & AD83:AD87)
    const mapMinat = {
      'Sales': 'O83', 'Teknologi Informasi (IT)': 'O84', 'Production / Manufaktur': 'O85',
      'Service / Maintenance': 'O86', 'Engineering / Design': 'O87',
      'Front Office': 'AD83', 'Accounting / Pembukuan': 'AD84', 'General Affairs': 'AD85',
      'Administrasi': 'AD86', 'HRD': 'AD87'
    };
    (d.minat || []).forEach((m, idx) => {
      if (mapMinat[m]) flk.getRange(mapMinat[m]).setValue(idx + 1);
    });

    flk.getRange('N90').setValue(d.gaji_diinginkan);
    flk.getRange('I93').setValue(d.hobi);

    // Organisasi (Baris 98 - 99)
    (d.org || []).forEach((o, i) => {
      const r = 98 + i;
      if (r <= 99 && o.nama) {
        flk.getRange('B' + r).setValue(o.nama);
        flk.getRange('M' + r).setValue(o.tempat);
        flk.getRange('S' + r).setValue(o.jab);
        flk.getRange('AA' + r).setValue(o.thn);
      }
    });

    // Kenalan (Baris 103 - 104)
    (d.kenalan || []).forEach((kn, i) => {
      const r = 103 + i;
      if (r <= 104 && kn.nama) {
        flk.getRange('B' + r).setValue(kn.nama);
        flk.getRange('M' + r).setValue(kn.jab);
        flk.getRange('V' + r).setValue(kn.hub);
      }
    });

    // Psikotes (Baris 108 - 109)
    (d.psiko || []).forEach((ps, i) => {
      const r = 108 + i;
      if (r <= 109 && ps.tempat) {
        flk.getRange('B' + r).setValue(ps.tempat);
        flk.getRange('M' + r).setValue(ps.thn);
        flk.getRange('V' + r).setValue(ps.tujuan);
      }
    });

    flk.getRange('N111').setValue(d.waktu_luang);
    flk.getRange('A116').setValue(d.kelebihan);
    flk.getRange('A119').setValue(d.kekurangan);
    flk.getRange('A124').setValue(d.gambaran_diri);
    flk.getRange('P135').setValue(d.kota);
    flk.getRange('T135').setValue(new Date().toLocaleDateString('id-ID'));
    flk.getRange('R140').setValue(d.nama);
    insertSign(flk, 'R136');
  }

  // ---------------- B. ISI SHEET "No.Rekening" ----------------
  const rek = tempSS.getSheetByName('No.Rekening');
  if (rek) {
    rek.getRange('E14').setValue(d.nama);
    rek.getRange('E16').setValue(d.tempat_lahir + ', ' + d.tgl_lahir);
    rek.getRange('E18').setValue(d.agama);
    rek.getRange('D24').setValue("'" + d.bank_norek);
    rek.getRange('E24').setValue(d.bank_alamat);
    rek.getRange('F24').setValue(d.bank_pemilik);
    rek.getRange('F44').setValue(d.kota + ', ' + new Date().toLocaleDateString('id-ID'));
    rek.getRange('F49').setValue('( ' + d.nama + ' )');
    insertSign(rek, 'F45');
  }

  // ---------------- C. ISI SHEET "S Setor Uang (SLS,CASHIER)" ----------------
  const setor = tempSS.getSheetByName('S Setor Uang (SLS,CASHIER)');
  if (setor) {
    setor.getRange('C5').setValue(': ' + d.nama);
    setor.getRange('C6').setValue(': ' + (hrd.npk || ''));
    setor.getRange('C7').setValue(': ' + d.alamat);
    setor.getRange('C8').setValue(': ' + d.kota + ' ' + d.kodepos);
    setor.getRange('C9').setValue(': ' + d.posisi);
    setor.getRange('C10').setValue(': ' + d.ktp);
    setor.getRange('A20').setValue(d.kota + ', ' + new Date().toLocaleDateString('id-ID'));
    setor.getRange('A25').setValue('( ' + d.nama + ' )');
    insertSign(setor, 'A22');
  }

  // ---------------- D. ISI SHEET "DPA" ----------------
  const dpa = tempSS.getSheetByName('DPA');
  if (dpa) {
    const dob = (d.tgl_lahir || '--').split('-');
    const tglM = (hrd.tglmasuk || '--').split('-');
    dpa.getRange('E9').setValue('PT ASTRA INTERNATIONAL Tbk - DSO');
    dpa.getRange('AM10').setValue(hrd.kodept || '');
    dpa.getRange('E11').setValue(hrd.npk || '');
    dpa.getRange('E15').setValue(d.nama);
    dpa.getRange('E24').setValue(d.alamat + ', ' + d.kota + ' ' + d.kodepos);
    dpa.getRange('E28').setValue(d.email);
    dpa.getRange('E30').setValue("'" + d.telp_rumah);
    dpa.getRange('E32').setValue("'" + d.hp);
    if (dob.length === 3) {
      dpa.getRange('E34').setValue(dob[2]);
      dpa.getRange('K34').setValue(dob[1]);
      dpa.getRange('Q34').setValue(dob[0]);
    }
    if (tglM.length === 3) {
      dpa.getRange('E36').setValue(tglM[2]);
      dpa.getRange('K36').setValue(tglM[1]);
      dpa.getRange('Q36').setValue(tglM[0]);
    }
    dpa.getRange('G40').setValue(hrd.gapok || '');

    // Tabel Ahli Waris DPA (Baris 47 - 52)
    (d.waris || []).forEach((w, i) => {
      const r = 47 + i;
      if (r <= 52 && w.nama) {
        dpa.getRange('A' + r).setValue(i + 1);
        dpa.getRange('C' + r).setValue(w.nama);
        dpa.getRange('E' + r).setValue(w.ttl);
        dpa.getRange('N' + r).setValue(w.lp);
        dpa.getRange('V' + r).setValue(w.hub);
        dpa.getRange('AG' + r).setValue(w.pek);
      }
    });

    dpa.getRange('E58').setValue(d.bank_pemilik);
    dpa.getRange('E60').setValue(d.bank_nama + ' - ' + d.bank_alamat);
    dpa.getRange('E62').setValue("'" + d.bank_norek);
    dpa.getRange('AB70').setValue('( ' + d.nama + ' )');
    insertSign(dpa, 'AB66');
  }

  // ---------------- E. ISI SHEET "PDK" ----------------
  const pdk = tempSS.getSheetByName('PDK');
  if (pdk) {
    if (hrd.noform) pdk.getRange('AE4').setValue(': ' + hrd.noform);
    pdk.getRange('N10').setValue(d.nama);
    pdk.getRange('N11').setValue(hrd.npk || '');
    pdk.getRange('N12').setValue(d.agama);
    pdk.getRange('N13').setValue(d.jk);
    pdk.getRange('N14').setValue("'" + d.ktp);
    pdk.getRange('N15').setValue(d.tempat_lahir + ', ' + d.tgl_lahir);
    pdk.getRange('N16').setValue(d.alamat + ', ' + d.kota + ' ' + d.kodepos);
    pdk.getRange('N18').setValue("'" + d.hp);

    if (d.status_nikah === 'Menikah') pdk.getRange('N25').setValue(d.tgl_nikah);
    if (d.status_nikah === 'Bercerai') pdk.getRange('N26').setValue(d.tgl_nikah);

    // Susunan Keluarga PDK (Baris 31 - 35)
    const gabungKeluarga = (d.keluarga_inti || []).filter(x => x.nama).concat((d.keluarga_ortu || []).filter(x => x.nama));
    gabungKeluarga.slice(0, 5).forEach((k, i) => {
      const r = 31 + i;
      pdk.getRange('D' + r).setValue(i + 1);
      pdk.getRange('F' + r).setValue(k.hub);
      pdk.getRange('N' + r).setValue(k.nama);
      pdk.getRange('X' + r).setValue(k.lp);
      pdk.getRange('Z' + r).setValue(k.ttl);
      pdk.getRange('AH' + r).setValue(k.pek);
    });

    // Data Ahli Waris PDK (Baris 40 - 44)
    (d.waris || []).forEach((w, i) => {
      const r = 40 + i;
      if (r <= 44 && w.nama) {
        pdk.getRange('D' + r).setValue(i + 1);
        pdk.getRange('F' + r).setValue(w.hub);
        pdk.getRange('N' + r).setValue(w.nama);
        pdk.getRange('X' + r).setValue(w.lp);
        pdk.getRange('Z' + r).setValue(w.ttl);
        pdk.getRange('AH' + r).setValue(w.pek);
      }
    });

    pdk.getRange('C50').setValue(d.kota + ', ' + new Date().toLocaleDateString('id-ID'));
    pdk.getRange('C52').setValue('( ' + d.nama + ' )');
    insertSign(pdk, 'C50');
  }

  SpreadsheetApp.flush();

  // ---------------- EXPORT TIAP SHEET JADI PDF ASLI & BUNGKUS .ZIP ----------------
  const sheetsToExport = ['List Kelengkapan', 'FLK', 'No.Rekening', 'S Setor Uang (SLS,CASHIER)', 'DPA', 'PDK', 'BPJS'];
  const pdfBlobs = [];
  const token = ScriptApp.getOAuthToken();
  const ssId = tempSS.getId();

  sheetsToExport.forEach((sName, idx) => {
    const sh = tempSS.getSheetByName(sName);
    if (!sh) return;
    // Skip Surat Setor Uang jika bukan Sales / Counter Sales / Kasir
    if (sName === 'S Setor Uang (SLS,CASHIER)' && !/sales|kasir|cashier/i.test(d.posisi)) return;

    const url = 'https://docs.google.com/spreadsheets/d/' + ssId + '/export?' +
      'format=pdf&size=A4&portrait=true&fitw=true&sheetnames=false&printtitle=false&pagenumbers=false&gridlines=false&fzr=false&gid=' + sh.getSheetId();
    const res = UrlFetchApp.fetch(url, { headers: { Authorization: 'Bearer ' + token } });
    const cleanName = sName.replace(/[^a-zA-Z0-9]/g, '_');
    pdfBlobs.push(res.getBlob().setName('0' + (idx + 1) + '_' + cleanName + '_' + d.nama.replace(/\s+/g, '_') + '.pdf'));
  });

  const zipBlob = Utilities.zip(pdfBlobs, 'PKB_DSO_' + d.nama.replace(/\s+/g, '_') + '.zip');
  const zipBase64 = Utilities.base64Encode(zipBlob.getBytes());

  // Hapus file temporary setelah selesai di-export
  DriveApp.getFileById(ssId).setTrashed(true);

  return ContentService.createTextOutput(JSON.stringify({
    status: 'success',
    filename: 'PKB_DSO_' + d.nama.replace(/\s+/g, '_') + '.zip',
    zipBase64: zipBase64
  })).setMimeType(ContentService.MimeType.JSON);
}
